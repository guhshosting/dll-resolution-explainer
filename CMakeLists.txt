cmake_minimum_required(VERSION 3.24)
project(dll-resolution-explainer VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and the MSVC ABI toolchain are required")
endif()
option(DRE_ANALYZE "Enable MSVC analysis" OFF)
option(DRE_ASAN "Enable address sanitizer" OFF)
add_library(dre STATIC src/pe.cpp src/files.cpp)
target_include_directories(dre PUBLIC src)
target_compile_features(dre PUBLIC cxx_std_17)
target_compile_definitions(dre PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(dre PUBLIC /W4 /permissive- /utf-8 /EHsc)
if(DRE_ANALYZE)
  target_compile_options(dre PUBLIC /analyze)
endif()
if(DRE_ASAN)
  target_compile_options(dre PUBLIC /fsanitize=address /Zi)
  target_link_options(dre PUBLIC /INCREMENTAL:NO)
  get_filename_component(DRE_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(DRE_ASAN_RUNTIME "${DRE_COMPILER_DIR}/clang_rt.asan_dynamic-x86_64.dll")
  else()
    set(DRE_ASAN_RUNTIME "${DRE_COMPILER_DIR}/clang_rt.asan_dynamic-i386.dll")
  endif()
  if(EXISTS "${DRE_ASAN_RUNTIME}")
    configure_file("${DRE_ASAN_RUNTIME}" "${CMAKE_CURRENT_BINARY_DIR}/" COPYONLY)
  endif()
endif()
add_executable(dll-resolution-explainer src/main.cpp src/version.rc)
target_link_libraries(dll-resolution-explainer PRIVATE dre advapi32)
install(TARGETS dll-resolution-explainer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE docs/schema.json DESTINATION share/dll-resolution-explainer)
include(CTest)
if(BUILD_TESTING)
  add_executable(parser-tests tests/parser.cpp)
  target_link_libraries(parser-tests PRIVATE dre)
  add_test(NAME parser-corpus COMMAND parser-tests)
  add_test(NAME cli-help COMMAND dll-resolution-explainer --help)
  add_test(NAME cli-version COMMAND dll-resolution-explainer --version)
  add_library(fixture-dependency SHARED tests/dependency.cpp)
  add_executable(fixture-target tests/target.cpp)
  target_link_libraries(fixture-target PRIVATE fixture-dependency delayimp)
  target_link_options(fixture-target PRIVATE /DELAYLOAD:fixture-dependency.dll)
  add_test(NAME fixture-inspection COMMAND dll-resolution-explainer --json $<TARGET_FILE:fixture-target>)
  set_tests_properties(parser-corpus cli-help cli-version fixture-inspection PROPERTIES TIMEOUT 60)
  find_program(DRE_PWSH pwsh)
  if(DRE_PWSH)
    add_test(NAME integration-json COMMAND "${DRE_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.ps1" -Tool $<TARGET_FILE:dll-resolution-explainer> -Fixture $<TARGET_FILE:fixture-target> -Dependency $<TARGET_FILE:fixture-dependency>)
    set_tests_properties(integration-json PROPERTIES TIMEOUT 60)
  endif()
endif()
